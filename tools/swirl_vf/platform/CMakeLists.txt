cmake_minimum_required(VERSION 3.20)
project(SwirlFontNativeProbes LANGUAGES C CXX)

add_executable(directwrite_probe directwrite_probe.cpp)
target_compile_features(directwrite_probe PRIVATE cxx_std_17)
if(WIN32)
  target_link_libraries(directwrite_probe PRIVATE dwrite d2d1 windowscodecs ole32 bcrypt)
  if(MSVC)
    target_compile_options(directwrite_probe PRIVATE /W4 /utf-8)
  endif()
endif()

add_executable(hvf_oracle hvf_oracle.c)
set_property(TARGET hvf_oracle PROPERTY C_STANDARD 11)
if(APPLE)
  target_link_libraries(hvf_oracle PRIVATE hvf)
  target_compile_options(hvf_oracle PRIVATE -Wall -Wextra -Werror -mmacosx-version-min=15.4)
  target_link_options(hvf_oracle PRIVATE -mmacosx-version-min=15.4)
endif()
