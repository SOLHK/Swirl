cmake_minimum_required(VERSION 3.20)
project(SwirlTargetAxisTransportProbe LANGUAGES CXX)
add_executable(target_axis_transport ../target_axis_transport.cpp)
target_compile_features(target_axis_transport PRIVATE cxx_std_17)
if(WIN32)
    target_compile_definitions(target_axis_transport PRIVATE _WIN32_WINNT=0x0A00 NTDDI_VERSION=0x0A000004)
    target_link_libraries(target_axis_transport PRIVATE dwrite bcrypt)
    if(MSVC)
        target_compile_options(target_axis_transport PRIVATE /W4 /EHsc /utf-8)
    endif()
endif()
