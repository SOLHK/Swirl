cmake_minimum_required(VERSION 3.20)
project(SwirlPrecisionRoundingTrace LANGUAGES CXX)
add_executable(precision_rounding_trace ../precision_rounding_trace.cpp)
target_compile_features(precision_rounding_trace PRIVATE cxx_std_17)
if(WIN32)
    target_compile_definitions(precision_rounding_trace PRIVATE _WIN32_WINNT=0x0A00 NTDDI_VERSION=0x0A000004)
    target_link_libraries(precision_rounding_trace PRIVATE dwrite bcrypt)
    if(MSVC)
        target_compile_options(precision_rounding_trace PRIVATE /W4 /EHsc /utf-8)
    endif()
endif()
